import mongoose from 'mongoose'

export const database = async (): Promise<void> => {
    try {
        await mongoose.connect('mongodb://localhost:27017/pluggskap_province_db')
        console.error('MongoDB successfully connected')
    } catch (error) {
        console.error('MongoDB connection error', error);
        process.exit(1);
    }
};
