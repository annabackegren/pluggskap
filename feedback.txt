USE pluggskap_db;

CREATE TABLE feedback (
    feedbackId INT AUTO_INCREMENT NOT NULL,
    feedbackMessage TEXT NOT NULL,
    feedbackTeacherId INT NOT NULL,
    feedbackStudentId INT NOT NULL,
    feedbackProvinceId VARCHAR(10) NOT NULL,
    feedbackCreatedAt TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (feedbackId),
    FOREIGN KEY (feedbackTeacherId) REFERENCES user(userId),
    FOREIGN KEY (feedbackStudentId) REFERENCES user(userId)
);
