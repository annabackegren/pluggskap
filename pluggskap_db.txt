CREATE DATABASE pluggskap_db;

USE pluggskap_db;

CREATE TABLE user(
    userId INT AUTO_INCREMENT NOT NULL,
    userType VARCHAR(20) NOT NULL,
    userName VARCHAR(20) UNIQUE NOT NULL,
    userPassword VARCHAR(255) NOT NULL,
    PRIMARY KEY (userId)
);

CREATE TABLE question(
    questionId INT AUTO_INCREMENT NOT NULL,
    questionText VARCHAR(255) NOT NULL,
    questionImg VARCHAR(255),
    questionAnswerC VARCHAR(50) NOT NULL,
    questionAnswerW1 VARCHAR(50) NOT NULL,
    questionAnswerW2 VARCHAR(50) NOT NULL,
    questionAnswerW3 VARCHAR(50) NOT NULL,
    questionProvinceId VARCHAR(10) NOT NULL,
    PRIMARY KEY (questionId)
);

CREATE TABLE result(
    resultId INT AUTO_INCREMENT NOT NULL,
    resultScore INT NOT NULL,
    resultUserId INT NOT NULL,
    resultProvinceId VARCHAR(20) NOT NULL,
    PRIMARY KEY (resultId),
    FOREIGN KEY (resultUserId) REFERENCES user(userId)
);
